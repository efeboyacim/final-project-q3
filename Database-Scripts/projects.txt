-- Table: public.projects

-- DROP TABLE IF EXISTS public.projects;

CREATE TABLE IF NOT EXISTS public.projects
(
    id integer NOT NULL DEFAULT nextval('projects_id_seq'::regclass),
    name character varying(200) COLLATE pg_catalog."default" NOT NULL,
    description text COLLATE pg_catalog."default",
    owner_id integer NOT NULL,
    CONSTRAINT projects_pkey PRIMARY KEY (id),
    CONSTRAINT projects_owner_id_fkey FOREIGN KEY (owner_id)
        REFERENCES public.users (id) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE CASCADE
)

TABLESPACE pg_default;

ALTER TABLE IF EXISTS public.projects
    OWNER to postgres;
-- Index: ix_projects_id

-- DROP INDEX IF EXISTS public.ix_projects_id;

CREATE INDEX IF NOT EXISTS ix_projects_id
    ON public.projects USING btree
    (id ASC NULLS LAST)
    TABLESPACE pg_default;
-- Index: ix_projects_name

-- DROP INDEX IF EXISTS public.ix_projects_name;

CREATE INDEX IF NOT EXISTS ix_projects_name
    ON public.projects USING btree
    (name COLLATE pg_catalog."default" ASC NULLS LAST)
    TABLESPACE pg_default;
